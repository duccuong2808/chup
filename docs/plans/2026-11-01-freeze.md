```rust
//! Hình học chọn và chỉnh vùng trên overlay Freeze (SPEC 14.3): 8 handle, di chuyển, resize.
//! Toạ độ là physical px trong overlay của 1 màn hình. Module thuần, unit test được.

use crate::coords::Rect;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Grip {
    Move,
    N,
    NE,
    E,
    SE,
    S,
    SW,
    W,
    NW,
}

/// 8 handle là ô vuông cạnh `size`, tâm đặt ở 4 góc và trung điểm 4 cạnh.
/// Góc xếp trước cạnh, để `hit` ưu tiên góc khi vùng nhỏ làm các handle chồng nhau.
pub fn handles(r: Rect, size: i32) -> [(Grip, Rect); 8] {
    let (l, t, rt, b) = (r.x, r.y, r.right(), r.bottom());
    let (cx, cy) = (l + (rt - l) / 2, t + (b - t) / 2);
    let sq = |x: i32, y: i32| Rect::new(x - size / 2, y - size / 2, size as u32, size as u32);
    [
        (Grip::NW, sq(l, t)),
        (Grip::NE, sq(rt, t)),
        (Grip::SW, sq(l, b)),
        (Grip::SE, sq(rt, b)),
        (Grip::N, sq(cx, t)),
        (Grip::S, sq(cx, b)),
        (Grip::W, sq(l, cy)),
        (Grip::E, sq(rt, cy)),
    ]
}

/// Điểm `p` nằm trên handle nào, trong vùng (`Move`), hay ở ngoài (`None`).
pub fn hit(r: Rect, p: (i32, i32), size: i32) -> Option<Grip> {
    handles(r, size)
        .iter()
        .find(|(_, h)| h.contains_point(p.0, p.1))
        .map(|(g, _)| *g)
        .or_else(|| r.contains_point(p.0, p.1).then_some(Grip::Move))
}
```

```rust
/// Vùng mới khi kéo `grip` từ `from` tới `to`. Luôn tính từ vùng lúc nhấn chuột (`r`),
/// không cộng dồn qua từng lần di chuột.
/// - `Move`: giữ nguyên kích thước, kẹp trong `bounds`.
/// - Resize: kéo qua cạnh đối diện thì vùng lật chiều, tối thiểu 1×1, không vượt `bounds`.
pub fn apply(r: Rect, grip: Grip, from: (i32, i32), to: (i32, i32), bounds: Rect) -> Rect {
    let (dx, dy) = (to.0 - from.0, to.1 - from.1);
    if grip == Grip::Move {
        let x = (r.x + dx).clamp(bounds.x, (bounds.right() - r.w as i32).max(bounds.x));
        let y = (r.y + dy).clamp(bounds.y, (bounds.bottom() - r.h as i32).max(bounds.y));
        return Rect::new(x, y, r.w, r.h);
    }
    let (mut l, mut t, mut rt, mut b) = (r.x, r.y, r.right(), r.bottom());
    let cx = |v: i32| v.clamp(bounds.x, bounds.right());
    let cy = |v: i32| v.clamp(bounds.y, bounds.bottom());
    if matches!(grip, Grip::W | Grip::NW | Grip::SW) {
        l = cx(l + dx);
    }
    if matches!(grip, Grip::E | Grip::NE | Grip::SE) {
        rt = cx(rt + dx);
    }
    if matches!(grip, Grip::N | Grip::NW | Grip::NE) {
        t = cy(t + dy);
    }
    if matches!(grip, Grip::S | Grip::SW | Grip::SE) {
        b = cy(b + dy);
    }
    let mut out = Rect::from_points((l, t), (rt, b));
    if out.w == 0 {
        out.w = 1;
        out.x = out.x.min(bounds.right() - 1);
    }
    if out.h == 0 {
        out.h = 1;
        out.y = out.y.min(bounds.bottom() - 1);
    }
    out
}
```

```rust
//! Pixel cho overlay Freeze (SPEC 14.3, 14.4): chép ảnh đóng băng 1:1 vào DIB, phần ngoài vùng
//! chọn tối 40%; crop luôn từ `RgbaImage` gốc.

use image::RgbaImage;

use crate::coords::Rect;

/// 1 pixel RGBA -> BGRA premultiplied, alpha 255 (mọi chỗ trên overlay nhận chuột).
/// `dim`: nhân 154/256 ≈ 0,6 (tối 40%).
#[inline]
pub fn to_bgra(px: [u8; 4], dim: bool) -> u32 {
    let k = |c: u8| if dim { (c as u32 * 154) >> 8 } else { c as u32 };
    0xFF00_0000 | (k(px[0]) << 16) | (k(px[1]) << 8) | k(px[2])
}

/// Vẽ phần `dirty` của DIB `dst` (rộng `dst_w`, toạ độ overlay) từ `img`.
/// Trong `sel` giữ nguyên màu, ngoài `sel` tô tối. Phần DIB không có ảnh (ảnh nhỏ hơn màn hình) tô đen.
pub fn compose(img: &RgbaImage, dst: &mut [u32], dst_w: usize, dirty: Rect, sel: Option<Rect>) {
    let dst_h = dst.len() / dst_w.max(1);
    let Some(d) = dirty.intersect(&Rect::new(0, 0, dst_w as u32, dst_h as u32)) else { return };
    let (iw, ih) = (img.width() as i32, img.height() as i32);
    let raw = img.as_raw();
    for y in d.y..d.bottom() {
        let row = &mut dst[y as usize * dst_w..(y as usize + 1) * dst_w];
        // Đoạn sáng của hàng này, nếu hàng cắt qua vùng chọn.
        let (s0, s1) = match sel {
            Some(s) if y >= s.y && y < s.bottom() => (s.x, s.right()),
            _ => (0, 0),
        };
        for x in d.x..d.right() {
            row[x as usize] = if x < iw && y < ih {
                let i = (y as usize * iw as usize + x as usize) * 4;
                to_bgra([raw[i], raw[i + 1], raw[i + 2], raw[i + 3]], !(s0..s1).contains(&x))
            } else {
                0xFF00_0000
            };
        }
    }
}

/// Cắt `rect` (physical px toàn desktop) từ ảnh đóng băng của màn hình đặt tại `mon`.
/// `None` nếu vùng không cắt qua ảnh.
pub fn crop(img: &RgbaImage, mon: Rect, rect: Rect) -> Option<RgbaImage> {
    let img_r = Rect::new(mon.x, mon.y, img.width(), img.height());
    let r = rect.intersect(&img_r)?;
    Some(image::imageops::crop_imm(img, (r.x - mon.x) as u32, (r.y - mon.y) as u32, r.w, r.h).to_image())
}
```

```rust
/// Chụp mọi màn hình song song (Freeze, SPEC 14.3). Trả về rect physical của từng màn hình kèm ảnh.
/// Mỗi thread tự lấy `Monitor` của mình, không chia sẻ đối tượng xcap giữa các thread.
pub fn grab_monitors() -> Result<Vec<(Rect, Result<RgbaImage, String>)>, String> {
    let mons = xcap::Monitor::all().map_err(|e| format!("Không liệt kê được màn hình: {e}"))?;
    let rects: Vec<Rect> = mons
        .iter()
        .filter_map(|m| Some(Rect::new(m.x().ok()?, m.y().ok()?, m.width().ok()?, m.height().ok()?)))
        .collect();
    drop(mons);
    Ok(std::thread::scope(|s| {
        let jobs: Vec<_> = rects
            .iter()
            .map(|r| {
                s.spawn(move || {
                    let (cx, cy) = r.center();
                    xcap::Monitor::from_point(cx, cy)
                        .and_then(|m| m.capture_image())
                        .map_err(|e| format!("Chụp màn hình thất bại: {e}"))
                })
            })
            .collect();
        rects
            .iter()
            .zip(jobs)
            .map(|(r, j)| (*r, j.join().unwrap_or_else(|_| Err("Thread chụp bị panic".into()))))
            .collect()
    }))
}
```

```rust
/// 1 nút trên thanh. `primary`: nền màu nhấn. `active`: icon màu nhấn (nút Mờ khi đang Faded).
#[derive(Clone, Copy)]
pub struct BarButton {
    pub icon: Icon,
    pub label: &'static str,
    pub primary: bool,
    pub active: bool,
}

fn button_width(font: Option<&Font>, k: f32, b: &BarButton, icons_only: bool) -> f32 {
    if icons_only || b.label.is_empty() {
        28.0 * k
    } else {
        let tw = font.map(|f| f.measure(b.label, 1000.0).0).unwrap_or(30.0 * k);
        (10.0 + 16.0 + 6.0) * k + tw + 10.0 * k
    }
}

/// Chiều rộng thanh (physical) cho bộ nút `items`.
pub fn bar_width(font: Option<&Font>, k: f32, items: &[BarButton], icons_only: bool) -> f32 {
    4.0 * k * (items.len() as f32 + 1.0) + items.iter().map(|b| button_width(font, k, b, icons_only)).sum::<f32>()
}

/// Vẽ thanh nút căn phải trong khung `size` (giữa `Canvas::begin` và `end`). Trả về rect từng nút.
/// Bố cục (DIP): thanh cao 36, nút cao 28, đệm 4, cách mép phải 2.
#[allow(clippy::too_many_arguments)]
pub fn paint_bar(
    c: &Canvas,
    font: Option<&Font>,
    theme: &Theme,
    k: f32,
    size: (i32, i32),
    items: &[BarButton],
    icons_only: bool,
    hover: Option<usize>,
    pressed: Option<usize>,
) -> Vec<Rect> {
    let (w, h) = (size.0 as f32, size.1 as f32);
    let (bh, bh_btn, pad) = (36.0 * k, 28.0 * k, 4.0 * k);
    let bar_w = bar_width(font, k, items, icons_only);
    let (bar_x, bar_y) = (w - bar_w - 2.0 * k, (h - bh) / 2.0);
    let by = bar_y + (bh - bh_btn) / 2.0;
    let mut x = bar_x + pad;
    let mut rects = Vec::with_capacity(items.len());
    for b in items {
        let bw = button_width(font, k, b, icons_only);
        rects.push(Rect::new(x as i32, by as i32, bw as u32, bh_btn as u32));
        x += bw + pad;
    }

    let bar = rectf(bar_x, bar_y, bar_w, bh);
    c.shadow(bar, 10.0 * k, 3.0 * k, 1.0 * k, theme.shadow);
    c.fill_round(bar, 10.0 * k, theme.surface);
    c.stroke_round(bar, 10.0 * k, 1.0, theme.line);
    for (i, (b, r)) in items.iter().zip(&rects).enumerate() {
        let rf = rectf(r.x as f32, r.y as f32, r.w as f32, r.h as f32);
        let down = if pressed == Some(i) { 1.0 * k } else { 0.0 };
        let hovered = hover == Some(i);
        let (bg, fg) = if b.primary {
            (Some(if hovered { theme.accent_strong } else { theme.accent }), Color::rgba(255, 255, 255, 1.0))
        } else if b.active {
            (hovered.then_some(theme.surface2), theme.accent)
        } else {
            (hovered.then_some(theme.surface2), theme.text)
        };
        if let Some(bg) = bg {
            c.fill_round(rf, 6.0 * k, bg);
        }
        let icon_sz = 16.0 * k;
        let iy = rf.top + (r.h as f32 - icon_sz) / 2.0 + down;
        if icons_only || b.label.is_empty() {
            c.icon(b.icon, rf.left + (r.w as f32 - icon_sz) / 2.0, iy, icon_sz, fg);
        } else {
            c.icon(b.icon, rf.left + 10.0 * k, iy, icon_sz, fg);
            if let Some(f) = font {
                let tx = rf.left + (10.0 + 16.0 + 6.0) * k;
                c.text(b.label, f, rectf(tx, rf.top + down, rf.right - tx, rf.bottom - rf.top), fg);
            }
        }
    }
    rects
}
```

```rust
    BarButton { icon: Icon::Copy, label: "Copy", primary: false, active: false },
    BarButton { icon: Icon::Close, label: "", primary: false, active: false },
];
/// Hành động của từng nút; `None` = huỷ.
const ACTIONS: [Option<FreezeAction>; 3] = [Some(FreezeAction::Save), Some(FreezeAction::Copy), None];

pub(super) enum Drag {
    Idle,
    New((i32, i32)),
    Adjust { grip: Grip, from: (i32, i32), rect0: Rect },
    Button(usize),
}

pub(super) struct Frozen {
    /// `None`: màn hình này chụp hỏng, chỉ phủ tối kèm thông báo, không cho chọn.
    img: Option<RgbaImage>,
    sel: Option<Rect>,
    drag: Drag,
    hint: Option<Sprite>,
    bar: Option<Bar>,
    handle: i32,
}

/// Toolbar vẽ sẵn vào sprite riêng, khi cần mới blend vào DIB của overlay.
struct Bar {
    canvas: Canvas, // thả trước surface
    surface: Surface,
    font: Option<Font>,
    k: f32,
    /// Chỗ đặt trên overlay (tính lại mỗi khi vùng đổi).
    at: Rect,
    buttons: Vec<Rect>,
    hover: Option<usize>,
    pressed: Option<usize>,
}

impl Bar {
    fn new(k: f32) -> Option<Bar> {
        let font = Font::new(13.0 * k, false, Align::Leading, false);
        let w = (bar_width(font.as_ref(), k, &BUTTONS, false) + 4.0 * k).ceil() as i32;
        let h = (BAR_H_DIP as f32 * k).ceil() as i32;
        let surface = Surface::new(w, h)?;
        let canvas = Canvas::new(&surface)?;
        let mut b = Bar { canvas, surface, font, k, at: Rect::default(), buttons: vec![], hover: None, pressed: None };
        b.render();
        Some(b)
    }

    fn render(&mut self) {
        let theme = Theme::current();
        self.canvas.begin(Color::rgba(0, 0, 0, 0.0));
        let size = (self.surface.w, self.surface.h);
        self.buttons =
            paint_bar(&self.canvas, self.font.as_ref(), &theme, self.k, size, &BUTTONS, false, self.hover, self.pressed);
        self.canvas.end();
    }

    fn hit(&self, p: (i32, i32)) -> Option<usize> {
        let (x, y) = (p.0 - self.at.x, p.1 - self.at.y);
        self.buttons.iter().position(|r| r.contains_point(x, y))
    }
}

impl Frozen {
    /// Toolbar chỉ hiện khi đã có vùng và không kéo gì.
    fn bar_visible(&self) -> bool {
        self.sel.is_some() && matches!(self.drag, Drag::Idle | Drag::Button(_))
    }

    fn bar_hit(&self, p: (i32, i32)) -> Option<usize> {
        self.bar.as_ref().filter(|_| self.bar_visible()).and_then(|b| b.hit(p))
    }
}

/// Thay đổi cần vẽ lại sau một sự kiện chuột.
enum Change {
    None,
    /// Chỉ toolbar đổi (hover, nhấn): vẽ lại đúng chỗ toolbar.
    Bar,
    /// Vùng chọn hoặc trạng thái kéo đổi: vẽ lại phần cũ ∪ phần mới.
    Sel,
}

/// Phần overlay có thể đổi theo vùng chọn: vùng kèm handle, nhãn, toolbar, gợi ý.
fn extent(st: &State) -> Rect {
    let Some(f) = st.frozen.as_ref() else { return Rect::default() };
    let mut r = f.sel.map(|s| s.expand(f.handle + 2)).unwrap_or_default();
    let bar = f.bar.as_ref().filter(|_| f.bar_visible()).map(|b| b.at);
    for o in [st.label_at, bar, f.hint.as_ref().map(|h| h.at)].into_iter().flatten() {
        r = r.union(&o);
    }
    r
}

impl Overlay {
    /// Hiện overlay ở chế độ Freeze. `img = None`: màn hình này chụp hỏng.
    pub fn show_frozen(&self, focus: bool, img: Option<RgbaImage>) {
        let (w, h) = (self.mon.rect.w as i32, self.mon.rect.h as i32);
        let t0 = std::time::Instant::now();
        let k = self.mon.scale as f32;
        let text = if img.is_some() { HINT_FROZEN } else { HINT_FAILED };
        let white = Color::rgba(255, 255, 255, 1.0);
        let hint = pill(text, 13.0 * k, 12.0 * k, 6.0 * k, 999.0, Color::rgba(0, 0, 0, 0.7), white).map(|s| {
            let at = Rect::new((w - s.w) / 2, self.s(24.0), s.w as u32, s.h as u32);
            Sprite { s, at }
        });
        let bar = Bar::new(k);
        {
            let mut st = self.st.borrow_mut();
            if st.surface.is_none() {
                st.surface = Surface::new(w, h);
            }
            if st.surface.is_none() {
                log::error!("Không cấp được bộ nhớ cho overlay Freeze {w}×{h}");
                return;
            }
            st.start = None;
            st.sel = None;
            st.label_at = None;
            st.label_text.clear();
            st.frozen = Some(Frozen { img, sel: None, drag: Drag::Idle, hint, bar, handle: self.s(HANDLE_DIP) });
            super::paint(&mut st, Rect::new(0, 0, w as u32, h as u32));
        }
        let t_paint = t0.elapsed();
        if let Some(s) = self.st.borrow().surface.as_ref() {
            s.present(self.hwnd, (self.mon.rect.x, self.mon.rect.y), 255, None);
        }
        self.raise(focus);
        log::debug!("overlay freeze {} {w}×{h}: tô {t_paint:?}, tổng {:?}", self.mon.key, t0.elapsed());
    }

    /// Lấy ảnh đóng băng ra để crop (Lưu/Copy). Overlay vẫn hiện tới khi `hide`.
    pub fn take_frozen(&self) -> Option<RgbaImage> {
        self.st.borrow_mut().frozen.as_mut()?.img.take()
    }

    /// Người dùng bắt đầu chọn ở màn hình khác: bỏ vùng chọn ở màn hình này.
    pub fn clear_frozen_selection(&self) {
        self.frozen_update(|f| {
            if f.sel.take().is_some() {
                f.drag = Drag::Idle;
                Change::Sel
            } else {
                Change::None
            }
        });
    }

    fn bounds(&self) -> Rect {
        Rect::new(0, 0, self.mon.rect.w, self.mon.rect.h)
    }

    /// Chạy `f` trên state Freeze, đặt lại nhãn và toolbar, rồi vẽ và đẩy phần đã đổi.
    /// Di chuột chỉ để hover thì chỉ vẽ lại toolbar: compose cả vùng lớn mỗi lần chuột di sẽ giật ở 4K.
    fn frozen_update(&self, f: impl FnOnce(&mut Frozen) -> Change) {
        let mut st = self.st.borrow_mut();
        let before = extent(&st);
        let Some(fr) = st.frozen.as_mut() else { return };
        let dirty = match f(fr) {
            Change::None => return,
            Change::Bar => match fr.bar.as_ref() {
                Some(b) => b.at,
                None => return,
            },
            Change::Sel => {
                let sel = fr.sel;
                let dragging = matches!(fr.drag, Drag::New(_) | Drag::Adjust { .. });
                let gap = fr.handle / 2 + self.s(BAR_GAP_DIP);
                st.label_at = match sel {
                    Some(s) if dragging => self.place_label(&mut st, s),
                    _ => None,
                };
                // Toolbar đặt trên toàn màn hình (được đè lên taskbar trong ảnh đóng băng).
                let bounds = self.bounds();
                if let (Some(s), Some(b)) = (sel, st.frozen.as_mut().and_then(|f| f.bar.as_mut())) {
                    let size = (b.surface.w as u32, b.surface.h as u32);
                    let (x, y) = place_toolbar(&s, size, &bounds, gap);
                    b.at = Rect::new(x, y, size.0, size.1);
                }
                before.union(&extent(&st))
            }
        };
        super::paint(&mut st, dirty);
        if let Some(s) = st.surface.as_ref() {
            s.present(self.hwnd, (self.mon.rect.x, self.mon.rect.y), 255, Some(dirty));
        }
    }
}
```

```rust
    pub(super) fn frozen_down(&self, p: (i32, i32), dbl: bool) {
        let hit_px = self.s(HANDLE_HIT_DIP);
        let mut started_new = false;
        let mut save = false;
        self.frozen_update(|f| {
            if f.img.is_none() {
                return Change::None;
            }
            if let Some(i) = f.bar_hit(p) {
                f.drag = Drag::Button(i);
                if let Some(b) = f.bar.as_mut() {
                    b.pressed = Some(i);
                    b.render();
                }
                return Change::Bar;
            }
            match f.sel.and_then(|s| selection::hit(s, p, hit_px)) {
                Some(Grip::Move) if dbl => {
                    save = true;
                    Change::None
                }
                Some(grip) => {
                    f.drag = Drag::Adjust { grip, from: p, rect0: f.sel.unwrap_or_default() };
                    Change::Sel
                }
                None => {
                    f.drag = Drag::New(p);
                    f.sel = None;
                    started_new = true;
                    Change::Sel
                }
            }
        });
        if started_new {
            emit(Action::FreezeSelecting(self.hwnd.0 as isize));
        }
        if save {
            self.frozen_confirm(Some(FreezeAction::Save));
        }
    }

    pub(super) fn frozen_move(&self, p: (i32, i32)) {
        let b = self.bounds();
        let p = (p.0.clamp(0, b.right()), p.1.clamp(0, b.bottom()));
        self.frozen_update(|f| match f.drag {
            Drag::New(start) => {
                let r = Rect::from_points(start, p);
                f.sel = (!r.is_empty()).then_some(r);
                Change::Sel
            }
            Drag::Adjust { grip, from, rect0 } => {
                f.sel = Some(selection::apply(rect0, grip, from, p, b));
                Change::Sel
            }
            Drag::Idle => {
                let h = f.bar_hit(p);
                match f.bar.as_mut() {
                    Some(bar) if bar.hover != h => {
                        bar.hover = h;
                        bar.render();
                        Change::Bar
                    }
                    _ => Change::None,
                }
            }
            Drag::Button(_) => Change::None,
        });
    }

    pub(super) fn frozen_up(&self, p: (i32, i32)) {
        let mut clicked = None;
        self.frozen_update(|f| match std::mem::replace(&mut f.drag, Drag::Idle) {
            Drag::Button(i) => {
                if f.bar_hit(p) == Some(i) {
                    clicked = Some(i);
                }
                if let Some(b) = f.bar.as_mut() {
                    b.pressed = None;
                    b.render();
                }
                Change::Bar
            }
            Drag::New(_) => {
                // Giống Live: vùng nhỏ hơn 4×4 coi như chưa chọn.
                if f.sel.is_some_and(|s| s.w < 4 || s.h < 4) {
                    f.sel = None;
                }
                Change::Sel
            }
            Drag::Adjust { .. } => Change::Sel,
            Drag::Idle => Change::None,
        });
        if let Some(i) = clicked {
            self.frozen_confirm(ACTIONS[i]);
        }
    }

    /// `None` = huỷ. Lưu/Copy chỉ chạy khi đã có vùng.
    pub(super) fn frozen_confirm(&self, action: Option<FreezeAction>) {
        let sel = self.st.borrow().frozen.as_ref().and_then(|f| f.sel);
        match (action, sel) {
            (None, _) => emit(Action::FreezeCancel),
            (Some(action), Some(r)) => {
                emit(Action::FreezeConfirm { rect: r.offset(self.mon.rect.x, self.mon.rect.y), action })
            }
            (Some(_), None) => {}
        }
    }

    pub(super) fn frozen_cursor(&self, p: (i32, i32)) -> PCWSTR {
        let st = self.st.borrow();
        let Some(f) = st.frozen.as_ref() else { return IDC_CROSS };
        if f.bar_hit(p).is_some() {
            return IDC_HAND;
        }
        match f.sel.and_then(|s| selection::hit(s, p, self.s(HANDLE_HIT_DIP))) {
            Some(Grip::Move) => IDC_SIZEALL,
            Some(Grip::N | Grip::S) => IDC_SIZENS,
            Some(Grip::E | Grip::W) => IDC_SIZEWE,
            Some(Grip::NW | Grip::SE) => IDC_SIZENWSE,
            Some(Grip::NE | Grip::SW) => IDC_SIZENESW,
            None => IDC_CROSS,
        }
    }
```

```rust
/// Vẽ phần `dirty` ở chế độ Freeze: ảnh (tối ngoài vùng), viền, handle, nhãn, gợi ý, toolbar.
pub(super) fn paint_frozen(st: &mut State, dirty: Rect) {
    let State { surface, frozen, label, label_at, .. } = st;
    let (Some(s), Some(f)) = (surface.as_mut(), frozen.as_mut()) else { return };
    let sw = s.w as usize;
    match &f.img {
        Some(img) => freeze::compose(img, s.pixels(), sw, dirty, f.sel),
        None => s.fill(dirty, Color::rgba(0, 0, 0, 0.6).px()),
    }
    if let Some(sel) = f.sel {
        s.outline_outside(sel, 1, WHITE);
        if !matches!(f.drag, Drag::New(_)) {
            for (_, hr) in selection::handles(sel, f.handle) {
                if hr.intersect(&dirty).is_some() {
                    s.fill(hr.expand(1), DARK);
                    s.fill(hr, WHITE);
                }
            }
        }
    }
    if f.sel.is_none() && matches!(f.drag, Drag::Idle) {
        if let Some(h) = f.hint.as_mut() {
            if h.at.intersect(&dirty).is_some() {
                s.blend_from(&mut h.s, h.at.x, h.at.y);
            }
        }
    }
    if let (Some(p), Some(at)) = (label.as_mut(), label_at.as_ref()) {
        s.blend_part(&mut p.surface, (at.w as i32, at.h as i32), at.x, at.y);
    }
    let show_bar = f.bar_visible();
    if let Some(b) = f.bar.as_mut().filter(|_| show_bar) {
        if b.at.intersect(&dirty).is_some() {
            s.blend_from(&mut b.surface, b.at.x, b.at.y);
        }
    }
}
```

```rust
    fn start_freeze_grab(&mut self) {
        self.freeze_t0 = Some(std::time::Instant::now());
        std::thread::spawn(|| {
            let t0 = std::time::Instant::now();
            let r = capture::grab_monitors();
            log::info!("freeze: chụp mọi màn hình mất {:?}", t0.elapsed());
            emit(Action::FreezeGrabbed(r));
        });

    fn show_freeze_overlays(&mut self) {
        self.sync_overlays();
        let mut shots = std::mem::take(&mut self.frozen);
        let (cx, cy) = cursor_pos();
        let focus = self.overlays.iter().position(|o| o.mon.rect.contains_point(cx, cy)).unwrap_or(0);
        let order = (0..self.overlays.len()).filter(|i| *i != focus).chain(std::iter::once(focus));
        for i in order {
            let Some(o) = self.overlays.get(i) else { continue };
            // Ghép theo góc trên-trái: kích thước xcap báo có thể lệch vài px so với EnumDisplayMonitors.
            let img = shots
                .iter()
                .position(|(r, _)| (r.x, r.y) == (o.mon.rect.x, o.mon.rect.y))
                .and_then(|p| shots.swap_remove(p).1.ok());
            o.show_frozen(i == focus, img);
        }
        if let Some(t0) = self.freeze_t0.take() {
            log::info!("freeze: từ nhấn hotkey tới lúc ảnh hiện {:?}", t0.elapsed());
        }
    }

    fn freeze_output(&mut self, rect: Rect, save: bool) {
        let Some((mon, img)) = self.picked.take() else {
            log::warn!("freeze: không có ảnh để crop");
            return;
        };
        let opts = self.model.persist_options();
        std::thread::spawn(move || {
            let shot = chup_core::freeze::crop(&img, mon, rect);
            drop(img);
            let Some(shot) = shot else {
                emit(Action::Toast(Toast::error("Vùng chọn nằm ngoài ảnh")));
                return;

            if save {
                let t0 = std::time::Instant::now();
                let r = capture::persist_shot(&shot, &opts);
                log::info!("freeze: lưu ảnh mất {:?}", t0.elapsed());
                emit(Action::Persisted(r));
            } else {
                emit(Action::FreezeCopied(chup_core::clipboard::copy_image(&shot, None)));
            }
        });
```
